    76→function fmt(bytes: number): string {
    77→  if (bytes === 0) return '0 B'
    78→  const k = 1024, s = ['B', 'KB', 'MB', 'GB', 'TB']
    79→  const i = Math.floor(Math.log(bytes) / Math.log(k))
    80→  return `${(bytes / Math.pow(k, i)).toFixed(1)} ${s[i]}`
    81→}

    83→function getftype(m: string): string {
    84→  if (m.startsWith('image/')) return 'Images'
    85→  if (m.startsWith('video/')) return 'Videos'
    86→  if (m.startsWith('audio/')) return 'Audio'
    87→  if (m.includes('pdf')) return 'PDFs'
    88→  if (m.includes('document') || m.includes('spreadsheet') || m.includes('presentation')) return 'Documents'
    89→  if (m.includes('zip') || m.includes('rar') || m.includes('tar')) return 'Archives'
    90→  return 'Other Files'
    91→}
